cmake_minimum_required(VERSION 3.16)
project(PokeWavePlugin LANGUAGES CXX)
find_package(Threads REQUIRED)
set(TS3_SDK_DIR "" CACHE PATH "Path to a checkout of teamspeak/ts3client-pluginsdk")
if(NOT EXISTS "${TS3_SDK_DIR}/include/ts3_functions.h")
  message(FATAL_ERROR "TS3_SDK_DIR must contain include/ts3_functions.h")
endif()
add_library(pokewave SHARED src/pokewave_plugin.cpp)
target_compile_features(pokewave PRIVATE cxx_std_17)
target_include_directories(pokewave PRIVATE "${TS3_SDK_DIR}/include" "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_link_libraries(pokewave PRIVATE Threads::Threads)
set_target_properties(pokewave PROPERTIES PREFIX "" OUTPUT_NAME "pokewave")
if(WIN32)
  target_compile_definitions(pokewave PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE _CRT_SECURE_NO_WARNINGS)
  target_link_libraries(pokewave PRIVATE comctl32)
else()
  target_compile_options(pokewave PRIVATE -fvisibility=hidden)
endif()
